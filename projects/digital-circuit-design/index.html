03Troubleshooting

Troubleshooting

Overview

Six-slot lot (2 accessible, 2 regular, 1 EV, 1 compact). Priorities for six cases by vehicle type and disability, plus unexpected events: a slot freeing up, an assigned slot taken, shaded slots in summer.

02Approach

System design

  • Controller (8-state FSM) and Datapath (9 blocks) organized with an ASMD chart to divide the work
  • Basic operation written in Verilog first, verified with Verilog and PSpice before building
  • After proposal feedback, added a FAIL signal and a re-search from the current position

Shortest-Time Route Navigation

Semester

Team

01Objective

Overview

Finds the fastest route among bus, transfer, and walking routes with an ASMD-based FSM and datapath.

02Approach

System Design and Circuits

System design

  • 15-state one-hot Controller, 16-bit one-hot current position (blocks 0 – 15)
  • Counters search up and down for the nearest stop and compare bus arrival times
  • Total times of bus-only, transfer, and walking routes compared, route stored in registers
  • Accident: leave the bus and search again. Congestion: bus weight replaced with the congestion value

Circuits

03Troubleshooting

Troubleshooting

  • Voltage drop across about 14 breadboards → buffers and several power supplies
  • Timing mismatch among clocks → bus, pedestrian, and delay clocks synchronized, clocks only in action states
  • Controller too large → small controllers built from counters inside states

module slot_controller_new (
 output reg clr, clr_e, clr_s, load_m, load_b, load_e, s_f, load_s, shift, up_d, w2, w3,
 input button, in, out, has, ev, can, fail, change, give_up, slot_ev, clk, reset
);
  
  reg has_e, stay;

  reg [7:0] state;
  parameter s0=8'b0000_0001, s1=8'b0000_0010, s2=8'b0000_0100, s3=8'b0000_1000, s4=8'b0001_0000, s5=8'b0010_0000, s6=8'b0100_0000, s7=8'b1000_0000;
  
  reg state2;
  parameter SS0=0, SSE=1;

  always @ (posedge clk, negedge reset) begin
    if (!reset)               state <= s0;
    else
      case (state)
        s0: if (button)       state <= s7;
            else            
              if (has_e)      state <= s1;
              else            state <= s0;
        s1: if (has)          state <= s4;
            else if (!ev)     state <= s2;
            else if (!stay)   state <= s0;
            else              state <= s2;
        s2: if (has)          state <= s4;
            else              state <= s3;
        s3: if (has)          state <= s4;
            else              state <= s6;
        s4: if (can)          state <= s5;
            else if (fail)    state <= s1;
            else if (change)  state <= s1;
            else              state <= s4;
        s5:                   state <= s0;
        s6: if (give_up)      state <= s0;
            else              state <= s1;
        s7: if (!in) begin
              if (out)        state <= s0;
              else            state <= s7;
            end
            else begin
              if (!ev)        state <= s1;
              else if (!stay) state <= s1;
              else            state <= s2;
            end
      endcase
  end

  always @ (state, stay, button, has_e, in, has, ev, can, fail, change, give_up) begin
    clr <= 1'b0;
    clr_s  <= 1'b0;
    load_m <= 1'b0;
    load_b <= 1'b0;
    load_e <= 1'b0;
    s_f <= 1'b0;
    load_s <= 1'b0;
    shift <= 1'b0;
    up_d <= 1'b0;
    w2 <= 1'b0;
    w3 <= 1'b0;
      case (state)
        s0: begin
                                            clr_s <= 1'b1;
            if (button)                     load_m <= 1'b1;
            else if (has_e)                 load_b <= 1'b1;
            else                            clr  <= 1'b1;
            end
        s1: if (has)                        load_s <= 1'b1;
            else if (ev&&(!stay)) 
            begin
                                            load_e <= 1'b1;
                                            clr <= 1'b1;
            end
        s2: begin
                                            w2 <= 1'b1;
            if (has)                        load_s <= 1'b1;
            end
        s3: begin
                                            w3 <= 1'b1;
            if (has)                        load_s <= 1'b1;
            end
        s4: begin
                                            shift <= 1'b1;
            if (can)                        up_d <= 1'b1;
            end
        s5: begin
                                            up_d <= 1'b1;
                                            clr <= 1'b1;
           	end
        s6: begin
                                            s_f <= 1'b1;
            if (give_up)                    clr <= 1'b1;
            end
        s7: if ((!in)&&out)                 clr <= 1'b1;
      endcase
  end

  always @ (posedge clk, negedge reset) begin
    if (!reset)                state2 <= SS0;
    else
      case (state2)
        SS0: if (load_e)        state2 <= SSE;
             else               state2 <= SS0;
        SSE: if (slot_ev)       state2 <= SSE;
             else
               if (state==s0)   state2 <= SS0;
               else             state2 <= SSE;
      endcase
  end

  always @ (state2, load_e, slot_ev, state) begin
    has_e <= 1'b0;
    stay <= 1'b0;
    clr_e <= 1'b0;
      case (state2)
        SS0: if (!load_e)                 clr_e <= 1'b0;
        SSE: begin
                                          stay <= 1'b1;
             if ((!slot_ev)&&(state==s0)) 
               begin
               has_e <= 1'b1;
               clr_e <= 1'b1;
               end
             end
      endcase
  end
  
endmodule

module slot_datapath_new (
  output button, in, out, has, ev, can, fail, change, give_up, slot_ev,
  output reg [5:0] slot_n, slot_s,
  input [5:0] slot_b,
  input clr, clr_e, clr_s, load_m, load_b, load_e, s_f, load_s, shift, up_d, w2, w3,
  input pg, ps, pe, dg, ds, de, mod, in_b, out_b, can_b, fail_b, give_up_b, clk, reset
);

  reg [2:0] REG, REGE;
  reg [2:0] car;
  parameter pg_n=3'b001, ps_n=3'b010, pe_n=3'b011, dg_n=3'b100, ds_n=3'b101, de_n=3'b110;
  reg [2:0] priority1, priority2, priority3, priority4, priority5, priority6;
  reg [2:0] pr2, pr3, pr4, pr5, pr6;
  reg [7:0] decoder;
  reg [7:0] MUX;
  reg [1:0] car_2, car_1, car_0;
  parameter s00=2'b00, s01=2'b01, s10=2'b10, s11=2'b11;
  reg [2:0] car_in;
  reg [5:0] en;
  reg [5:0] slot_c;
  
  wire [2:0] priority;
  wire [5:0] AND;
  wire [5:0] XOR;
  wire [5:0] pg_m, ps_m, pe_m, dg_m, ds_m, de_m;
  
  assign ev = (REG==pe_n)||(REG==de_n);
  assign has = |AND;
  assign slot_ev = slot_n[5];
  assign AND = MUX&(~slot_n);
  assign button = |car_in;
  assign priority = priority1|pr2|pr3|pr4|pr5|pr6;
  assign XOR = slot_n ^ slot_c;
  assign change = |XOR;
  assign in = in_b;
  assign out = out_b;
  assign can = can_b;
  assign fail = fail_b;
  assign give_up = give_up_b;
            //       F[5]               E[4]                                    D[3]                          C[2]                       B[1]                          A[0]
  assign pg_m = {        1'b0,                   (!w2)&(!w3), ((!mod)&(!w2)&(!w3))|(mod&w2),                          1'b0,                          1'b0,                          1'b0};
  assign ps_m = {        1'b0,                   (!w2)&(!w3), ((!mod)&(!w2)&(!w3))|(mod&w2), ((!mod)&(!w2)&(!w3))|(mod&w2),                          1'b0,                          1'b0};
  assign pe_m = { (!w2)&(!w3),                            w2,                            w3,                          1'b0,                          1'b0,                          1'b0};
  assign dg_m = {        1'b0, ((!mod)&w2)|(mod&(!w2)&(!w3)),          ((!mod)&w2)|(mod&w3),                          1'b0, ((!mod)&(!w2)&(!w3))|(mod&w2), ((!mod)&(!w2)&(!w3))|(mod&w2)};
  assign ds_m = {        1'b0, ((!mod)&w2)|(mod&(!w2)&(!w3)),          ((!mod)&w2)|(mod&w3),          ((!mod)&w2)|(mod&w3), ((!mod)&(!w2)&(!w3))|(mod&w2), ((!mod)&(!w2)&(!w3))|(mod&w2)};
  assign de_m = { (!w2)&(!w3),                            w3,                            w3,                          1'b0,                            w2,                            w2};
  
  always @ (pg, ps, pe, dg, ds, de) begin
                 car <= 3'b000;
    if (pg)      car <= pg_n;
    else if (ps) car <= ps_n;
    else if (pe) car <= pe_n;
    else if (dg) car <= dg_n;
    else if (ds) car <= ds_n;
    else if (de) car <= de_n;
    else         car <= 3'b000;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset) car_2 <= s00;
    else
      case (car_2)
        s00: if (car[2]) car_2 <= s01;
             else        car_2 <= s00;
        s01:             car_2 <= s10;
        s10: if (car[2]) car_2 <= s11;
             else        car_2 <= s00;
        s11: if (car[2]) car_2 <= s11;
             else        car_2 <= s00; 
      endcase
  end
  always @ (car_2) begin
    if ((car_2==s01)||(car_2==s10)) car_in[2] <= 1'b1;
    else                            car_in[2] <= 1'b0;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset) car_1 <= s00;
    else
      case (car_1)
        s00: if (car[1]) car_1 <= s01;
             else        car_1 <= s00;
        s01:             car_1 <= s10;
        s10: if (car[1]) car_1 <= s11;
             else        car_1 <= s00;
        s11: if (car[1]) car_1 <= s11;
             else        car_1 <= s00; 
      endcase
  end
  always @ (car_1) begin
    if ((car_1==s01)||(car_1==s10)) car_in[1] <= 1'b1;
    else                            car_in[1] <= 1'b0;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset) car_0 <= s00;
    else
      case (car_0)
        s00: if (car[0]) car_0 <= s01;
             else        car_0 <= s00;
        s01:             car_0 <= s10;
        s10: if (car[0]) car_0 <= s11;
             else        car_0 <= s00;
        s11: if (car[0]) car_0 <= s11;
             else        car_0 <= s00; 
      endcase
  end
  always @ (car_0) begin
    if ((car_0==s01)||(car_0==s10)) car_in[0] <= 1'b1;
    else                            car_in[0] <= 1'b0;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset)      REG <= 3'b000;
    else if (clr)    REG <= 3'b000;
    else if (load_m) REG <= load_m | car_in;
    else if (load_b) REG <= load_b | REGE;
    else             REG <= REG;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset)      REGE <= 3'b000;
    else if (clr_e)  REGE <= 3'b000;   
    else if (load_e) REGE <= REG;
    else             REGE <= REGE;
  end
  
  always @ (REG, mod, w2, w3) begin
    case (REG)
      pg_n: MUX <= pg_m;
      ps_n: MUX <= ps_m;
      pe_n: MUX <= pe_m;
      dg_n: MUX <= dg_m;
      ds_n: MUX <= ds_m;
      de_n: MUX <= de_m;

  end

    if (!reset)  en <= 6'b111_110;
    else

      else       en <= en;
  end

    if (en[0])             priority1 <= 3'b000;

    endcase
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset)      slot_s <= 6'b000_000;
    else if (clr_s)  slot_s <= 6'b000_000;
    else if (load_s) slot_s <= decoder[6:1];
    else             slot_s <= slot_s;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset)      slot_c <= 6'b000_000;
    else             slot_c <= slot_n;
  end
  
  always @ (posedge clk, negedge reset) begin
    if (!reset)      slot_n <= 6'b000_000;
    else if (up_d)   slot_n <= slot_n|slot_b|slot_s;
    else             slot_n <= slot_n|slot_b;
  end
  
endmodule
